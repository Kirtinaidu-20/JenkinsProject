pipeline{
    agent any

    stages{
        stage('Build'){
            steps{
                sh'mvn clean package'
            }
            
        
        }
        stage('Test'){
            steps{
                sh 'mvn test'
            }

       }
    }

    post{
        success{
            echo"pipeline successfull"
        }
        failure{
            echo "pipeline failed"
        }

    }

}
